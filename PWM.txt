
#include <LPC17xx.h>

unsigned int row, col;
unsigned long int row_bit, col_value;
unsigned char key;

unsigned int intensity;
unsigned int duty_values[4] = {3000, 7500, 15000, 22500};

unsigned char keypad[4][4] = {
    {'1', '2', '3', 'A'},
    {'4', '5', '6', 'B'},
    {'7', '8', '9', 'C'},
    {'*', '0', '#', 'D'}
};

void delay(void)
{
    unsigned int i;
    for (i = 0; i < 10000; i++);
}

unsigned char keypad_scan(void)
{
    for (row = 0; row < 4; row++)
    {
        // Make all rows LOW: P0.15-P0.18
        LPC_GPIO0->FIOCLR = (0x0F << 15);

        // Make current row HIGH
        row_bit = 1UL << (15 + row);
        LPC_GPIO0->FIOSET = row_bit;

        delay();

        // Read columns P0.19-P0.22
        col_value = (LPC_GPIO0->FIOPIN >> 19) & 0xF;

        if (col_value != 0)
        {
            if (col_value & 0x1)
                col = 0;
            else if (col_value & 0x2)
                col = 1;
            else if (col_value & 0x4)
                col = 2;
            else if (col_value & 0x8)
                col = 3;
            else
                continue;

            LPC_GPIO0->FIOCLR = (0x0F << 15);

            return keypad[row][col];
        }
    }

    LPC_GPIO0->FIOCLR = (0x0F << 15);
    return 0;
}

void pwm_init(void)
{
    LPC_SC->PCONP |= (1 << 6);

    // PWM1.4 on P1.23, Function 2
    LPC_PINCON->PINSEL3 &= ~(3 << 14);
    LPC_PINCON->PINSEL3 |=  (2 << 14);

    LPC_PWM1->PR  = 0;
    LPC_PWM1->PCR = (1 << 12);   // Enable PWM1.4, single edge
    LPC_PWM1->MCR = (1 << 1);    // Reset on MR0

    LPC_PWM1->MR0 = 30000;
    LPC_PWM1->MR4 = 3000;        // Initial intensity: 10%

    LPC_PWM1->LER = (1 << 0) | (1 << 4);

    LPC_PWM1->TCR = (1 << 1);    // Reset counter and prescaler
    LPC_PWM1->TCR = (1 << 0) | (1 << 3);
                                  // Enable PWM and counter
}

void set_intensity(unsigned char key)
{
    if (key == '0')
        intensity = 0;       // 10%
    else if (key == '1')
        intensity = 1;       // 25%
    else if (key == '2')
        intensity = 2;       // 50%
    else if (key == '3')
        intensity = 3;       // 75%
    else
        return;

    LPC_PWM1->MR4 = duty_values[intensity];
    LPC_PWM1->LER = (1 << 0) | (1 << 4);
}

int main(void)
{
    SystemInit();
    SystemCoreClockUpdate();

    // Configure P0.15-P0.22 as GPIO
    LPC_PINCON->PINSEL0 &= ~(0xFFFF0000);

    // Configure keypad rows P0.15-P0.18 as outputs
    LPC_GPIO0->FIODIR |= (0x0F << 15);

    // Configure keypad columns P0.19-P0.22 as inputs
    LPC_GPIO0->FIODIR &= ~(0x0F << 19);

    pwm_init();

    while (1)
    {
        key = keypad_scan();

        if (key != 0)
        {
            delay();

            if (keypad_scan() == key)
            {
                set_intensity(key);

                // Wait for key release
                while (keypad_scan() != 0)
                    delay();

                delay();
            }
        }
    }
}
