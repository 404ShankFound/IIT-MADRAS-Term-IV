#include <LPC17xx.h>

// Pin connections assumed:
// Seven-segment segments a-g, dp : P0.4 to P0.11 (output)
// Digit select / decoder inputs  : P1.23 to P1.26 (output)
// Four-digit multiplexed display
// Segment codes assumed active-HIGH, as in the sample code.
//
// A 4-bit ring counter rotates a single 1:
// 0001 -> 0010 -> 0100 -> 1000 -> 0001
// The 4-bit pattern is displayed as a decimal digit
// on the rightmost seven-segment position.

unsigned int i, j;
unsigned int number = 1;
unsigned int digit;

unsigned char seven_seg[16] = {
    0x3F, 0x06, 0x5B, 0x4F,
    0x66, 0x6D, 0x7D, 0x07,
    0x7F, 0x6F, 0x77, 0x7C,
    0x39, 0x5E, 0x79, 0x71
};

int main(void)
{
    SystemInit();
    SystemCoreClockUpdate();

    // Configure relevant pins for GPIO function.
    LPC_PINCON->PINSEL0 &= ~(0x03FF << 8);
    LPC_PINCON->PINSEL1 &= ~(0x03FF << 0);
    LPC_PINCON->PINSEL3 &= ~(0xFF << 14);

    // Segment outputs: P0.4-P0.11.
    LPC_GPIO0->FIODIR |= (0xFF << 4);

    // Digit decoder outputs: P1.23-P1.26.
    LPC_GPIO1->FIODIR |= (0x0F << 23);

    while (1)
    {
        // Display the current 4-bit ring-counter value.
        // Show the value as a decimal digit on the rightmost position.
        for (i = 0; i < 100; i++)
        {
            // Select rightmost digit (decoder value 0).
            LPC_GPIO1->FIOPIN = (0 << 23);

            // Display the current number on P0.4-P0.11.
            LPC_GPIO0->FIOPIN =
                (LPC_GPIO0->FIOPIN & ~(0xFF << 4))
                | ((unsigned int)seven_seg[number] << 4);

            // Multiplexing delay.
            for (j = 0; j < 1000; j++);
        }

        // Rotate the 1-bit left: 0001 -> 0010 -> 0100 -> 1000.
        number = number << 1;

        // Wrap around after 1000.
        if (number > 8)
            number = 1;

        // Delay between successive ring-counter states.
        for (j = 0; j < 50000; j++);
    }
}
